Zero Trust as Code

Automate your
journey to Zero Trust

Bring security up to DevOps speed with an API-first design
that easily integrates with your stack, no changes to
infrastructure needed. Just choose your IaC, choose your
VPC, and deploy.

Deploy withTerraform Deploy withPulumi Deploy withOur API
Terraform
Pulumi
API 
terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 4.0"
    }
    twingate = {
      source = "twingate/twingate"
    }
  }
}

variable "AWS_ACCESS_KEY_ID" {}
variable "AWS_SECRET_ACCESS_KEY" {}
variable "tg_api_key" {}
variable "tg_network" {}

# Configure the AWS Provider
provider "aws" {
  region     = "eu-west-1"
  access_key = var.AWS_ACCESS_KEY_ID
  secret_key = var.AWS_SECRET_ACCESS_KEY
}

# Configure Twingate Provider
provider "twingate" {
  api_token = var.tg_api_key
  network   = var.tg_network
}
